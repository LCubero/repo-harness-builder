```python
def generated_skill_replacements(
    installed_manifest: dict[str, Any],
    desired_manifest: dict[str, Any],
) -> dict[str, str]:
    """Map exact manifest-owned legacy skill files to fixed-name destinations."""
    project = installed_manifest.get("project")
    require(isinstance(project, dict) and isinstance(project.get("id"), str), "manifest ownership is unverifiable")
    project_id = project["id"]
    legacy_names = legacy_command_to_skill(project_id)
    canonical = installed_manifest.get("canonicalSkills")
    require(
        isinstance(canonical, dict) and set(canonical) == set(legacy_names.values()),
        "manifest must declare the exact legacy generated skills",
    )
    installed_paths = generated_skill_paths(installed_manifest)
    desired_paths = generated_skill_paths(desired_manifest)
    replacements: dict[str, str] = {}
    for command, source_name in legacy_names.items():
        destination_name = COMMAND_TO_SKILL[command]
        for source in sorted(path for path in installed_paths if f"/{source_name}/" in path):
            destination = source.replace(f"/{source_name}/", f"/{destination_name}/")
            require(destination in desired_paths, "migration destination is not manifest-owned")
            replacements[source] = destination
    require(set(replacements) == installed_paths, "migration must include every legacy generated skill path")
    require(set(replacements.values()) == desired_paths, "migration must include every fixed generated skill path")
    return replacements
```
